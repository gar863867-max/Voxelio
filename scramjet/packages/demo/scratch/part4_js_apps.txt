'use strict';

const AppRegistry = {
  apps: new Map(),

  register(appDef) {
    // appDef: { id, name, icon, category, defaultWidth, defaultHeight, singleton, component }
    this.apps.set(appDef.id, appDef);
  },

  open(appId, options = {}) {
    const appDef = this.apps.get(appId);
    if (!appDef) {
      console.error(`App ${appId} is not registered.`);
      return;
    }

    if (appDef.singleton) {
      const activeWinId = Array.from(QuantumOS.WindowManager.windows.keys()).find(id => id.startsWith(appId));
      if (activeWinId) {
        QuantumOS.WindowManager.focus(activeWinId);
        const win = QuantumOS.WindowManager.windows.get(activeWinId);
        if (appDef.component) {
          const bodyEl = win.element.querySelector('.window-body');
          appDef.component(bodyEl, options);
        }
        return win;
      }
    }

    const winId = appDef.singleton ? appId : `${appId}_${Math.random().toString(36).substr(2, 9)}`;
    const win = QuantumOS.WindowManager.createWindow({
      id: winId,
      title: appDef.name,
      icon: appDef.icon,
      w: options.w || appDef.defaultWidth || 640,
      h: options.h || appDef.defaultHeight || 440,
      content: ''
    });

    const bodyEl = win.element.querySelector('.window-body');
    if (appDef.component) {
      appDef.component(bodyEl, options);
    }

    if (typeof lucide !== 'undefined') {
      lucide.createIcons({ attrs: { class: 'lucide-icon' } });
    }

    return win;
  },

  getAll() {
    return Array.from(this.apps.values());
  },

  search(query) {
    if (!query) return this.getAll();
    const q = query.toLowerCase();
    return this.getAll().filter(app =>
      app.name.toLowerCase().includes(q) ||
      (app.category && app.category.toLowerCase().includes(q))
    );
  },

  isRunning(appId) {
    return Array.from(QuantumOS.WindowManager.windows.keys()).some(id => id === appId || id.startsWith(appId + '_'));
  }
};

window.AppRegistry = AppRegistry;

// Hook desktop/startmenu launchers to AppRegistry

  browser: { launch(url) { AppRegistry.open('browser', { url }); } },

  settings: { launch(cat) { AppRegistry.open('settings', { category: cat }); } },

  musicPlayer: { launch() { AppRegistry.open('musicPlayer'); } },
  gamesBrowser: { launch() { AppRegistry.open('gamesBrowser'); } }
};

          <div class="explorer-grid-container" style="flex:1; padding:16px; overflow-y:auto; position:relative; background:var(--bg2);">
            <div class="explorer-view-list" style="display:flex; flex-direction:column; gap:4px;"></div>

          </div>
        </div>
        <div style="display:flex; justify-content:space-between; align-items:center; padding:6px 12px; border-top:1px solid var(--border); background:var(--bg3); font-size:11px; color:var(--t3); flex-shrink:0;">
          <div class="status-left">0 items</div>
          <div>Virtual Disk Persistent</div>
        </div>
      </div>
    `;

    const gridContainer = container.querySelector('.explorer-grid-container');
    const viewGrid = container.querySelector('.explorer-view-grid');
    const viewList = container.querySelector('.explorer-view-list');
    const pathInput = container.querySelector('.explorer-path');
    const btnBack = container.querySelector('.btn-back');
    const btnForward = container.querySelector('.btn-forward');
    const btnUp = container.querySelector('.btn-up');
    const btnRefresh = container.querySelector('.btn-refresh');
    const btnNewFolder = container.querySelector('.btn-new-folder');
    const btnNewFile = container.querySelector('.btn-new-file');
    const btnToggleView = container.querySelector('.btn-toggle-view');
    const searchInput = container.querySelector('.explorer-search');
    const sidebarLinks = container.querySelectorAll('.sidebar-link');
    const statusLeft = container.querySelector('.status-left');

    function updateToolbar() {
      btnBack.style.opacity = historyStack.length > 0 ? '1' : '0.5';
      btnForward.style.opacity = forwardStack.length > 0 ? '1' : '0.5';
    }

    function navigateTo(path, pushToHistory = true) {
      if (pushToHistory && currentPath !== path) {
        historyStack.push(currentPath);
        forwardStack = [];
      }
      currentPath = path;
      pathInput.value = currentPath;
      updateToolbar();
      renderContents();
    }

    function renderContents() {
      sidebarLinks.forEach(link => {
        const linkPath = link.getAttribute('data-path');
        link.style.background = linkPath === currentPath ? 'rgba(255,255,255,0.06)' : 'transparent';
      });

      const items = QuantumOS.FileSystem.list(currentPath);
      const query = searchInput.value.toLowerCase();
      const filtered = items.filter(f => f.name.toLowerCase().includes(query));

      selectedItems.clear();
      statusLeft.innerText = `${filtered.length} items`;

      if (viewMode === 'grid') {
        viewList.style.display = 'none';
        viewGrid.style.display = 'grid';
        viewGrid.innerHTML = '';
        if (filtered.length === 0) {
          viewGrid.innerHTML = '<div style="grid-column:1/-1; color:var(--t3); font-size:12px; padding:8px;">Folder is empty</div>';
          return;
        }

        filtered.forEach(item => {
          const el = document.createElement('div');
          el.style.cssText = 'display:flex; flex-direction:column; align-items:center; gap:4px; padding:8px; border-radius:6px; cursor:pointer; text-align:center; font-size:11px; transition:background 0.15s;';
          const icon = item.type === 'directory' ? 'folder' : QuantumOS.DesktopManager.getFileIcon(item.name);
          el.innerHTML = `
            <div style="width:40px; height:40px; display:flex; align-items:center; justify-content:center; background:rgba(255,255,255,0.02); border:1px solid var(--border); border-radius:6px;">
              <i data-lucide="${icon}" style="width:20px; height:20px;"></i>
            </div>
            <span style="width:100%; text-overflow:ellipsis; overflow:hidden; white-space:nowrap; color:var(--t2);">${item.name}</span>
          `;

          el.setAttribute('draggable', 'true');
          el.addEventListener('dragstart', (e) => {
            e.dataTransfer.setData('text/plain', `path:${currentPath === '/' ? '' : currentPath}/${item.name}`);
          });

          if (item.type === 'directory') {
            el.addEventListener('dragover', (e) => e.preventDefault());
            el.addEventListener('drop', (e) => {
              e.preventDefault();
              const src = e.dataTransfer.getData('text/plain');
              if (src && src.startsWith('path:')) {
                const srcPath = src.replace('path:', '');
                const destPath = `${currentPath === '/' ? '' : currentPath}/${item.name}`;
                if (srcPath !== destPath && !srcPath.startsWith(destPath + '/')) {
                  QuantumOS.FileSystem.move(srcPath, destPath);
                  renderContents();
                }
              }
            });
          }

          el.addEventListener('click', (e) => {
            e.stopPropagation();
            if (e.ctrlKey) {
              if (selectedItems.has(item.name)) selectedItems.delete(item.name);
              else selectedItems.add(item.name);
            } else {
              selectedItems.clear();
              selectedItems.add(item.name);
              viewGrid.querySelectorAll('div').forEach(child => child.style.background = 'transparent');
            }
            el.style.background = selectedItems.has(item.name) ? 'rgba(255,255,255,0.06)' : 'transparent';
            statusLeft.innerText = `${filtered.length} items (${selectedItems.size} selected)`;
          });

          el.addEventListener('dblclick', (e) => {
            e.stopPropagation();
            const fullPath = (currentPath === '/' ? '' : currentPath) + '/' + item.name;
            if (item.type === 'directory') {
              navigateTo(fullPath);
            } else {
              const ext = item.name.split('.').pop().toLowerCase();
              if (['mp3', 'wav', 'ogg'].includes(ext)) {
                QuantumOS.apps.musicPlayer.launch(fullPath);
              } else if (['html', 'htm'].includes(ext)) {
                QuantumOS.apps.browser.launch(fullPath);
              } else {
                QuantumOS.apps.textEditor.launch(fullPath);
              }
            }
          });

          el.addEventListener('contextmenu', (e) => {
            e.stopPropagation();
            e.preventDefault();
            selectedItems.clear();
            selectedItems.add(item.name);
            showMenu(e.clientX, e.clientY, item);
          });

          viewGrid.appendChild(el);
        });
      } else {
        viewGrid.style.display = 'none';
        viewList.style.display = 'flex';
        viewList.innerHTML = '';
        if (filtered.length === 0) {
          viewList.innerHTML = '<div style="color:var(--t3); font-size:12px; padding:8px;">Folder is empty</div>';
          return;
        }

        filtered.forEach(item => {
          const el = document.createElement('div');
          el.style.cssText = 'display:flex; align-items:center; gap:12px; padding:6px 12px; border-radius:6px; cursor:pointer; font-size:12px; transition:background 0.15s; color:var(--t2);';
          const icon = item.type === 'directory' ? 'folder' : QuantumOS.DesktopManager.getFileIcon(item.name);
          el.innerHTML = `
            <i data-lucide="${icon}" style="width:16px; height:16px;"></i>
            <span style="flex:1; overflow:hidden; text-overflow:ellipsis; white-space:nowrap;">${item.name}</span>
            <span style="color:var(--t3); width:100px;">${item.type}</span>
          `;

          el.addEventListener('click', (e) => {
            e.stopPropagation();
            selectedItems.clear();
            selectedItems.add(item.name);
            viewList.querySelectorAll('div').forEach(child => child.style.background = 'transparent');
            el.style.background = 'rgba(255,255,255,0.06)';
          });

          el.addEventListener('dblclick', () => {
            const fullPath = (currentPath === '/' ? '' : currentPath) + '/' + item.name;
            if (item.type === 'directory') {
              navigateTo(fullPath);
            } else {
              QuantumOS.apps.textEditor.launch(fullPath);
            }
          });

          el.addEventListener('contextmenu', (e) => {
            e.stopPropagation();
            e.preventDefault();
            showMenu(e.clientX, e.clientY, item);
          });

          viewList.appendChild(el);
        });
      }

      if (typeof lucide !== 'undefined') {
        lucide.createIcons({ attrs: { class: 'lucide-icon' } });
      }
    }

    function showMenu(clientX, clientY, item = null) {
      const oldMenu = document.getElementById('explorer-ctx-menu');
      if (oldMenu) oldMenu.remove();

      const menu = document.createElement('div');
      menu.id = 'explorer-ctx-menu';
      menu.style.cssText = `position:fixed; left:${clientX}px; top:${clientY}px; z-index:9999; background:rgba(15,15,15,0.95); border:1px solid var(--border); border-radius:8px; padding:4px 0; min-width:140px; display:flex; flex-direction:column; box-shadow:0 10px 25px rgba(0,0,0,0.5);`;

      const itemPath = item ? (currentPath === '/' ? '' : currentPath) + '/' + item.name : '';

      if (item) {
        menu.innerHTML = `
          <button class="ctx-btn" id="ctx-open" style="padding:6px 12px; text-align:left; font-size:12px; color:var(--t1); display:flex; align-items:center; gap:8px; width:100%;"><i data-lucide="play" style="width:12px; height:12px;"></i>Open</button>
          <button class="ctx-btn" id="ctx-copy" style="padding:6px 12px; text-align:left; font-size:12px; color:var(--t1); display:flex; align-items:center; gap:8px; width:100%;"><i data-lucide="copy" style="width:12px; height:12px;"></i>Copy</button>
          <button class="ctx-btn" id="ctx-cut" style="padding:6px 12px; text-align:left; font-size:12px; color:var(--t1); display:flex; align-items:center; gap:8px; width:100%;"><i data-lucide="scissors" style="width:12px; height:12px;"></i>Cut</button>
          <button class="ctx-btn" id="ctx-rename" style="padding:6px 12px; text-align:left; font-size:12px; color:var(--t1); display:flex; align-items:center; gap:8px; width:100%;"><i data-lucide="pencil" style="width:12px; height:12px;"></i>Rename</button>
          <hr style="border:none; border-top:1px solid var(--border); margin:4px 0;"/>
          <button class="ctx-btn" id="ctx-delete" style="padding:6px 12px; text-align:left; font-size:12px; color:#ff5f57; display:flex; align-items:center; gap:8px; width:100%;"><i data-lucide="trash-2" style="width:12px; height:12px;"></i>Delete</button>
        `;
        document.body.appendChild(menu);

        menu.querySelector('#ctx-open').addEventListener('click', () => {
          menu.remove();
          if (item.type === 'directory') navigateTo(itemPath);
          else QuantumOS.apps.textEditor.launch(itemPath);
        });
        menu.querySelector('#ctx-copy').addEventListener('click', () => {
          menu.remove();
          QuantumOS.state.clipboard = { type: 'copy', path: itemPath };
        });
        menu.querySelector('#ctx-cut').addEventListener('click', () => {
          menu.remove();
          QuantumOS.state.clipboard = { type: 'cut', path: itemPath };
        });
        menu.querySelector('#ctx-rename').addEventListener('click', () => {
          menu.remove();
          const name = prompt("Rename to:", item.name);
          if (name && name !== item.name) {
            QuantumOS.FileSystem.rename(itemPath, name);
            renderContents();
            if (currentPath === '/Desktop') QuantumOS.DesktopManager.renderIcons();
          }
        });
        menu.querySelector('#ctx-delete').addEventListener('click', () => {
          menu.remove();
          if (currentPath === '/Trash') QuantumOS.FileSystem.delete(itemPath);
          else QuantumOS.FileSystem.trash(itemPath);
          renderContents();
          if (currentPath === '/Desktop') QuantumOS.DesktopManager.renderIcons();
        });
      } else {
        menu.innerHTML = `
          <button class="ctx-btn" id="ctx-newfolder" style="padding:6px 12px; text-align:left; font-size:12px; color:var(--t1); display:flex; align-items:center; gap:8px; width:100%;"><i data-lucide="folder-plus" style="width:12px; height:12px;"></i>New Folder</button>
          <button class="ctx-btn" id="ctx-newfile" style="padding:6px 12px; text-align:left; font-size:12px; color:var(--t1); display:flex; align-items:center; gap:8px; width:100%;"><i data-lucide="file-plus" style="width:12px; height:12px;"></i>New File</button>
          <button class="ctx-btn" id="ctx-paste" style="padding:6px 12px; text-align:left; font-size:12px; color:var(--t1); display:flex; align-items:center; gap:8px; width:100%; opacity:${QuantumOS.state.clipboard ? 1 : 0.4}"><i data-lucide="clipboard" style="width:12px; height:12px;"></i>Paste</button>
        `;
        document.body.appendChild(menu);

        menu.querySelector('#ctx-newfolder').addEventListener('click', () => {
          menu.remove();
          createFolder();
        });
        menu.querySelector('#ctx-newfile').addEventListener('click', () => {
          menu.remove();
          createFile();
        });
        menu.querySelector('#ctx-paste').addEventListener('click', () => {
          menu.remove();
          if (QuantumOS.state.clipboard) {
            const clip = QuantumOS.state.clipboard;
            if (clip.type === 'copy') QuantumOS.FileSystem.copy(clip.path, currentPath);
            else if (clip.type === 'cut') {
              QuantumOS.FileSystem.move(clip.path, currentPath);
              QuantumOS.state.clipboard = null;
            }
            renderContents();
            if (currentPath === '/Desktop') QuantumOS.DesktopManager.renderIcons();
          }
        });
      }

      if (typeof lucide !== 'undefined') lucide.createIcons({ attrs: { class: 'lucide-icon' } });

      const clear = () => {
        menu.remove();
        document.removeEventListener('mousedown', clear);
      };
      setTimeout(() => document.addEventListener('mousedown', clear), 50);
    }

    function createFolder() {
      let name = 'New Folder', idx = 1;
      while (QuantumOS.FileSystem.get((currentPath === '/' ? '' : currentPath) + '/' + name)) {
        name = `New Folder (${idx++})`;
      }
      QuantumOS.FileSystem.createFolder(currentPath, name);
      renderContents();
      if (currentPath === '/Desktop') QuantumOS.DesktopManager.renderIcons();
    }

    function createFile() {
      let name = 'Untitled.txt', idx = 1;
      while (QuantumOS.FileSystem.get((currentPath === '/' ? '' : currentPath) + '/' + name)) {
        name = `Untitled (${idx++}).txt`;
      }
      QuantumOS.FileSystem.createFile(currentPath, name, '');
      renderContents();
      if (currentPath === '/Desktop') QuantumOS.DesktopManager.renderIcons();
    }

    // Wiring up toolbar and sidebars
    btnBack.addEventListener('click', () => {
      if (historyStack.length > 0) {
        forwardStack.push(currentPath);
        navigateTo(historyStack.pop(), false);
      }
    });

    btnForward.addEventListener('click', () => {
      if (forwardStack.length > 0) {
        navigateTo(forwardStack.pop(), true);
      }
    });

    btnUp.addEventListener('click', () => {
      if (currentPath === '/' || currentPath === '') return;
      const parts = currentPath.split('/').filter(p => p);
      parts.pop();
      navigateTo('/' + parts.join('/'));
    });

    btnRefresh.addEventListener('click', renderContents);
    btnNewFolder.addEventListener('click', createFolder);
    btnNewFile.addEventListener('click', createFile);

    btnToggleView.addEventListener('click', () => {
      viewMode = viewMode === 'grid' ? 'list' : 'grid';
      btnToggleView.innerHTML = viewMode === 'grid' ? '<i data-lucide="layout-grid" style="width:14px; height:14px;"></i>' : '<i data-lucide="list" style="width:14px; height:14px;"></i>';
      if (typeof lucide !== 'undefined') lucide.createIcons({ attrs: { class: 'lucide-icon' } });
      renderContents();
    });

    pathInput.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') {
        const val = pathInput.value.trim();
        const node = QuantumOS.FileSystem.get(val);
        if (node && node.type === 'directory') navigateTo(val);
        else {
          QuantumOS.NotificationSystem.showToast('Invalid directory');
          pathInput.value = currentPath;
        }
      }
    });

    searchInput.addEventListener('input', renderContents);

    sidebarLinks.forEach(link => {
      link.addEventListener('click', () => navigateTo(link.getAttribute('data-path')));
    });

    gridContainer.addEventListener('contextmenu', (e) => {
      if (e.target === gridContainer || e.target === viewGrid || e.target === viewList) {
        e.preventDefault();
        showMenu(e.clientX, e.clientY);
      }
    });

    navigateTo(currentPath, false);
  }
});

  id: 'browser',
  name: 'Quantum Browser',
  icon: 'globe',
  category: 'Productivity',
  defaultWidth: 900,
  defaultHeight: 600,
  singleton: false,
  component: function(container, options) {
    let tabs = [];
    let activeTabId = null;
    let bookmarks = JSON.parse(localStorage.getItem('browser_bookmarks') || '[]');
    let history = JSON.parse(localStorage.getItem('browser_history') || '[]');

    if (window.LibcurlClient && !window.$scramjetController && window.ScramjetController) {
      try {

      try { return new URL(url).hostname; } catch (e) { return url; }

      tabs = tabs.filter(t => t.id !== tabId);
      const iframe = container.querySelector(`#iframe_${tabId}`);
      if (iframe) iframe.remove();

      if (tabs.length === 0) createTab();
      else {
        if (activeTabId === tabId) switchTab(tabs[tabs.length - 1].id);
        else renderTabs();
      }

    function navigateActiveTab(url) {
      const activeTab = tabs.find(t => t.id === activeTabId);
      if (!activeTab) return;

      let parsed = url.trim();
      if (!parsed.startsWith('http://') && !parsed.startsWith('https://')) {
        if (parsed.includes('.') && !parsed.includes(' ')) parsed = 'https://' + parsed;
        else parsed = 'https://google.com/search?q=' + encodeURIComponent(parsed);
      }

      activeTab.url = parsed;
      urlInput.value = parsed;
      addHistory(parsed);
      updateBookmarkButton(parsed);

      const iframe = container.querySelector(`#iframe_${activeTabId}`);
      if (iframe) {
        let finalUrl = parsed;
        if (window.$scramjetController) {
          try { finalUrl = window.$scramjetController.encodeUrl(parsed); } catch (e) {}
        }
        iframe.src = finalUrl;
      }

      activeTab.history = activeTab.history.slice(0, activeTab.historyIndex + 1);
      activeTab.history.push(parsed);
      activeTab.historyIndex = activeTab.history.length - 1;
    }

        }
      }

        tab.url = tab.history[tab.historyIndex];
        urlInput.value = tab.url;
        const iframe = container.querySelector(`#iframe_${activeTabId}`);
        if (iframe) {
          let final = tab.url;
          if (window.$scramjetController) final = window.$scramjetController.encodeUrl(tab.url);
          iframe.src = final;
        }
      }
    });

    btnNavReload.addEventListener('click', () => {
      const iframe = container.querySelector(`#iframe_${activeTabId}`);
      if (iframe) {
        const src = iframe.src;
        iframe.src = 'about:blank';
        setTimeout(() => iframe.src = src, 50);
      }
    });

    function toggleSidebar(type) {
      const isVisible = sidebarOverlay.style.display === 'flex';
      const isSameType = sidebarTitle.innerText === type;
      if (isVisible && isSameType) sidebarOverlay.style.display = 'none';
      else {
        sidebarOverlay.style.display = 'flex';
        sidebarTitle.innerText = type;
        renderSidebarList(type);
      }

    function renderSidebarList(type) {
      sidebarList.innerHTML = '';
      const list = type === 'Bookmarks' ? bookmarks : history;
      if (list.length === 0) {
        sidebarList.innerHTML = `<div style="color:var(--t3); padding:8px;">No records</div>`;
        return;

        el.style.cssText = 'display:flex; justify-content:space-between; align-items:center; padding:6px; border-radius:6px; cursor:pointer; background:rgba(255,255,255,0.01); transition:background 0.2s; color:var(--t2);';

        `;

        el.addEventListener('click', (e) => {
          if (e.target.closest('.delete-url-item')) {
            e.stopPropagation();
            if (type === 'Bookmarks') {
              bookmarks = bookmarks.filter(b => b !== url);
              localStorage.setItem('browser_bookmarks', JSON.stringify(bookmarks));
            } else {
              history = history.filter(h => h !== url);
              localStorage.setItem('browser_history', JSON.stringify(history));
            }
            renderSidebarList(type);
          } else {
            navigateActiveTab(url);
            sidebarOverlay.style.display = 'none';
          }
        });
        sidebarList.appendChild(el);
      });

      if (typeof lucide !== 'undefined') lucide.createIcons({ attrs: { class: 'lucide-icon' } });
    }

    btnShowBookmarks.addEventListener('click', () => toggleSidebar('Bookmarks'));
    btnShowHistory.addEventListener('click', () => toggleSidebar('History'));
    btnCloseSidebar.addEventListener('click', () => sidebarOverlay.style.display = 'none');

    createTab(options.url || 'https://wikipedia.org');
  }

          const payload = active.messages.map(m => {
            if (m.image) return { role: m.role, content: [{ type: 'text', text: m.content }, { type: 'image_url', image_url: { url: m.image } }] };
            return { role: m.role, content: m.content };
          });

          const res = await fetch("https://openrouter.ai/api/v1/chat/completions", {
            method: "POST",
            headers: { "Authorization": `Bearer ${key}`, "Content-Type": "application/json" },
            body: JSON.stringify({ model: selectedModel, messages: payload, stream: true })
          });

          thinking.remove();

          const bubble = document.createElement('div');
          bubble.style.cssText = 'max-width:80%; padding:10px 14px; border-radius:12px; font-size:13px; line-height:1.5; word-wrap:break-word; align-self:flex-start; background:var(--glass-bg); border:1px solid var(--glass-border); color:var(--t1);';
          messagesContainer.appendChild(bubble);

          const reader = res.body.getReader();
          const decoder = new TextDecoder();
          let full = '';

          while (true) {
            const { done, value } = await reader.read();
            if (done) break;
            const chunks = decoder.decode(value).split('\n');
            for (const chunk of chunks) {
              if (chunk.includes('[DONE]')) continue;
              if (chunk.startsWith('data: ')) {
                try {
                  const data = JSON.parse(chunk.slice(6));
                  full += data.choices[0]?.delta?.content || '';
                  bubble.innerHTML = markdown(full);
                  messagesContainer.scrollTop = messagesContainer.scrollHeight;
                } catch (e) {}
              }
            }
          }

          active.messages.push({ role: 'assistant', content: full });
          localStorage.setItem('ai_conversations', JSON.stringify(conversations));
        } catch (e) {
          thinking.remove();
          mockReply("Error connecting to OpenRouter. Please try again.");
        }
      } else {
        setTimeout(() => {
          thinking.remove();
          mockReply("OpenRouter key not detected. Set key in System settings under AI.");
        }, 1000);
      }
    }

    function mockReply(reply) {
      const active = conversations.find(c => c.id === currentConversationId);
      if (active) {
        active.messages.push({ role: 'assistant', content: reply });
        localStorage.setItem('ai_conversations', JSON.stringify(conversations));
      }
      renderMessages();
    }

    btnSend.addEventListener('click', sendMessage);
    textarea.addEventListener('keydown', (e) => {
      if (e.key === 'Enter' && !e.shiftKey) {
        e.preventDefault();
        sendMessage();
      }
    });

    modelSelect.value = selectedModel;
    modelSelect.addEventListener('change', (e) => {
      selectedModel = e.target.value;
      QuantumOS.SettingsEngine.set('ai.openRouterModel', selectedModel);
    });
    btnNewChat.addEventListener('click', createNewConversation);

    textarea.addEventListener('paste', (e) => {
      const items = e.clipboardData.items;
      for (const item of items) {
        if (item.type.indexOf('image') !== -1) {
          const blob = item.getAsFile();
          const reader = new FileReader();
          reader.onload = (ev) => {
            currentPastedImage = ev.target.result;
            imgPreview.src = currentPastedImage;
            imagePreviewBar.style.display = 'flex';
          };
          reader.readAsDataURL(blob);
          e.preventDefault();
        }
      }
    });

    btnClearImage.addEventListener('click', () => {
      currentPastedImage = null;
      imagePreviewBar.style.display = 'none';
    });

    renderConversations();
    renderMessages();
  }

AppRegistry.register({
  id: 'terminal',
  name: 'Console Terminal',
  icon: 'terminal',
  category: 'System',
  defaultWidth: 700,
  defaultHeight: 450,
  singleton: false,
  component: function(container, options) {
    let currentDir = '/Desktop';
    let history = [];
    let historyIndex = -1;

    container.innerHTML = `
      <div style="display:flex; flex-direction:column; height:100%; color:#00ff00; font-family:var(--font-mono); background:#050505; padding:12px; overflow:hidden; font-size:12px; line-height:1.4;">
        <div class="term-scroll" style="flex:1; overflow-y:auto; display:flex; flex-direction:column; gap:4px; margin-bottom:8px; white-space:pre-wrap;">Quantum OS Shell [v1.0.0]
(c) 2026 Quantum Corporation.

Type "help" for instructions.

</div>
        <div style="display:flex; align-items:center; flex-shrink:0;">
          <span class="prompt-text" style="color:#ffffff; font-weight:600; margin-right:6px;">quantum@quantum-os:${currentDir}$</span>
          <input type="text" class="term-input" style="flex:1; background:transparent; border:none; color:#00ff00; font-family:inherit; font-size:inherit; outline:none; caret-color:#00ff00;" autocomplete="off" />
        </div>
      </div>
    `;

    const scroll = container.querySelector('.term-scroll');
    const promptSpan = container.querySelector('.prompt-text');
    const input = container.querySelector('.term-input');

    const commands = {
      help: () => `Commands: help, ls, cd, pwd, mkdir, touch, rm, cat, echo, clear, whoami, date, uname, neofetch`,
      ls: (args) => {
        let t = args[0] || currentDir;
        if (!t.startsWith('/')) t = (currentDir === '/' ? '' : currentDir) + '/' + t;
        t = t.replace(/\/+/g, '/');
        const node = QuantumOS.FileSystem.get(t);
        if (!node) return `ls: directory not found: ${args[0]}`;
        const items = QuantumOS.FileSystem.list(t);
        return items.map(item => item.type === 'directory' ? `[DIR]  ${item.name}` : `[FILE] ${item.name}`).join('\n') || 'Empty';
      },
      cd: (args) => {
        let t = args[0] || '/Desktop';
        if (t === '..') {
          const parts = currentDir.split('/').filter(p => p);
          parts.pop();
          t = '/' + parts.join('/');
        } else if (!t.startsWith('/')) {
          t = (currentDir === '/' ? '' : currentDir) + '/' + t;
        }
        t = t.replace(/\/+/g, '/');
        const node = QuantumOS.FileSystem.get(t);
        if (node && node.type === 'directory') {
          currentDir = t;
          promptSpan.innerText = `quantum@quantum-os:${currentDir}$`;
          return '';
        }
        return `cd: not a directory: ${args[0]}`;
      },
      pwd: () => currentDir,
      mkdir: (args) => {
        if (!args[0]) return 'mkdir: name required';
        if (QuantumOS.FileSystem.createFolder(currentDir, args[0])) {
          if (currentDir === '/Desktop') QuantumOS.DesktopManager.renderIcons();
          return `Created folder ${args[0]}`;
        }
        return 'mkdir: failed (exists)';
      },
      touch: (args) => {
        if (!args[0]) return 'touch: name required';
        if (QuantumOS.FileSystem.createFile(currentDir, args[0], '')) {
          if (currentDir === '/Desktop') QuantumOS.DesktopManager.renderIcons();
          return `Created file ${args[0]}`;
        }
        return 'touch: failed (exists)';
      },
      rm: (args) => {
        if (!args[0]) return 'rm: file/folder required';
        const path = (currentDir === '/' ? '' : currentDir) + '/' + args[0];
        if (QuantumOS.FileSystem.delete(path)) {
          if (currentDir === '/Desktop') QuantumOS.DesktopManager.renderIcons();
          return `Deleted ${args[0]}`;
        }
        return `rm: failed to delete ${args[0]}`;
      },
      cat: (args) => {
        if (!args[0]) return 'cat: filename required';
        const path = (currentDir === '/' ? '' : currentDir) + '/' + args[0];
        const node = QuantumOS.FileSystem.get(path);
        if (node && node.type === 'file') return node.content || '';
        return `cat: ${args[0]}: file not found`;
      },
      echo: (args) => {
        if (args.includes('>')) {
          const idx = args.indexOf('>');
          const content = args.slice(0, idx).join(' ').replace(/^"(.*)"$/, '$1');
          const file = args[idx + 1];
          if (!file) return 'echo: invalid redirection';
          const path = (currentDir === '/' ? '' : currentDir) + '/' + file;
          const node = QuantumOS.FileSystem.get(path);
          if (node && node.type === 'file') {
            node.content = content;
            QuantumOS.FileSystem.save();
          } else {
            QuantumOS.FileSystem.createFile(currentDir, file, content);
          }
          if (currentDir === '/Desktop') QuantumOS.DesktopManager.renderIcons();
          return `Redirected to ${file}`;
        }
        return args.join(' ').replace(/^"(.*)"$/, '$1');
      },
      clear: () => { scroll.innerHTML = ''; return ''; },
      whoami: () => QuantumOS.state.currentUser || 'user',
      date: () => new Date().toString(),
      uname: () => `Quantum OS kernel 1.0.0`,
      neofetch: () => `    .::::.      quantum@quantum-os
   .::::::.     ----------------
   ::::::::     OS: Quantum OS 1.0.0
  ::::::::::    Shell: Virtual JS
  .::::::::.    Uptime: ${(performance.now()/1000).toFixed(0)}s
   .::::..      Display: Web Canvas`
    };

    function log(promptText, result = '') {
      if (promptText) scroll.innerText += promptText + '\n';
      if (result) scroll.innerText += result + '\n';
      scroll.scrollTop = scroll.scrollHeight;
    }

    input.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') {
        const val = input.value.trim();
        input.value = '';
        if (!val) return;

        history.push(val);
        historyIndex = history.length;

        const parts = val.split(' ');
        const op = parts[0];
        const args = parts.slice(1);

        const promptText = `quantum@quantum-os:${currentDir}$ ${val}`;

        if (commands[op]) {
          const res = commands[op](args);
          log(promptText, res);
        } else {
          log(promptText, `shell: command not found: ${op}`);
        }
      } else if (e.key === 'ArrowUp') {
        e.preventDefault();
        if (history.length > 0 && historyIndex > 0) {
          historyIndex--;
          input.value = history[historyIndex];
        }
      } else if (e.key === 'ArrowDown') {
        e.preventDefault();
        if (history.length > 0 && historyIndex < history.length - 1) {
          historyIndex++;
          input.value = history[historyIndex];
        } else {
          historyIndex = history.length;
          input.value = '';
        }
      } else if (e.key === 'Tab') {
        e.preventDefault();
        const val = input.value.trim();
        const parts = val.split(' ');
        const op = parts[0];

        if (parts.length === 1) {
          const matches = Object.keys(commands).filter(c => c.startsWith(op));
          if (matches.length === 1) input.value = matches[0];
          else if (matches.length > 1) log(`\n${matches.join('  ')}`);
        } else {
          const list = QuantumOS.FileSystem.list(currentDir);
          const term = parts[parts.length - 1];
          const matches = list.filter(item => item.name.startsWith(term));
          if (matches.length === 1) {
            parts[parts.length - 1] = matches[0].name;
            input.value = parts.join(' ');
          } else if (matches.length > 1) {
            log(`\n${matches.map(m => m.name).join('  ')}`);
          }
        }
      }
    });

    container.addEventListener('click', () => input.focus());
  }
});

          expr = expr.slice(0, -1);
          output.value = expr || '0';

          formula.innerText = expr + ' =';

            .replace(/tan\(/g, 'Math.tan(')

        QuantumOS.FileSystem.createFile('/Desktop', name, textarea.value);
      } else {
        const file = QuantumOS.FileSystem.get(filePath);
        if (file) {
          file.content = textarea.value;
          QuantumOS.FileSystem.save();
        }
      }
      QuantumOS.DesktopManager.renderIcons();
      QuantumOS.NotificationSystem.showToast('File saved.');
      const parts = filePath.split('/');
      const win = container.closest('.window');
      if (win) win.querySelector('.window-title span').innerText = `Text Editor - ${parts[parts.length-1]}`;
    });

    textarea.addEventListener('input', update);
    textarea.addEventListener('scroll', () => gutter.scrollTop = textarea.scrollTop);
    textarea.addEventListener('keyup', update);
    textarea.addEventListener('click', update);

    if (filePath) load(filePath);
    else update();
  }
});

AppRegistry.register({
  id: 'musicPlayer',
  name: 'Music Player',
  icon: 'music',
  category: 'Entertainment',
  defaultWidth: 700,
  defaultHeight: 440,
  singleton: false,
  component: function(container, options) {
    let state = false;
    let volume = 50;
    let loop = false;
    let shuffle = false;
    let activeIdx = 0;

    const tracks = [
      { name: 'Ambient Dreams', artist: 'Synth Wave', note: 'A4', dur: 120 },
      { name: 'Quantum Core Beat', artist: 'Oscillator', note: 'C5', dur: 90 },
      { name: 'Monochrome Drive', artist: 'Frequency Mod', note: 'E4', dur: 150 }
    ];

    let audioCtx = null;
    let osc = null;
    let gainNode = null;
    let analyserNode = null;
    let animId = null;
    let elapsed = 0;
    let timer = null;

    container.innerHTML = `
      <div style="display:flex; height:100%; color:var(--t1); font-family:var(--font-sans); background:var(--bg2);">
        <aside style="width:180px; border-right:1px solid var(--border); background:var(--bg); padding:12px; display:flex; flex-direction:column; gap:8px; flex-shrink:0;">
          <div style="font-size:10px; text-transform:uppercase; color:var(--t3); font-weight:700; padding:0 8px;">Playlist</div>
          <div class="list-container" style="display:flex; flex-direction:column; gap:4px; overflow-y:auto; flex:1;"></div>
        </aside>
        <main style="flex:1; padding:24px; display:flex; flex-direction:column; justify-content:space-between; align-items:center; gap:16px;">
          <div style="position:relative; width:130px; height:130px; border-radius:12px; background:var(--bg3); border:1px solid var(--border); display:flex; align-items:center; justify-content:center; overflow:hidden;">
            <canvas class="visualizer" width="130" height="130" style="position:absolute; inset:0; opacity:0.5; pointer-events:none;"></canvas>
            <i data-lucide="music" class="disc-icon" style="width:36px; height:36px; color:var(--t2); z-index:1;"></i>
          </div>
          <div style="text-align:center;">
            <strong class="title-el" style="font-size:14px; color:var(--t1);">Ambient Dreams</strong><br/>
            <span class="artist-el" style="font-size:11px; color:var(--t3);">Synth Wave</span>
          </div>
          <div style="width:100%; display:flex; align-items:center; gap:8px;">
            <span class="time-el" style="font-size:10px; color:var(--t3); width:30px;">0:00</span>
            <input type="range" class="seeker" min="0" max="100" value="0" style="flex:1; cursor:pointer;" />
            <span class="time-tot" style="font-size:10px; color:var(--t3); width:30px;">2:00</span>
          </div>
          <div style="display:flex; align-items:center; gap:16px;">
            <button class="btn-loop-track" style="color:var(--t3);"><i data-lucide="repeat" style="width:14px; height:14px;"></i></button>
            <button class="btn-prev-track"><i data-lucide="skip-back" style="width:16px; height:16px;"></i></button>
            <button class="btn-play-track" style="width:40px; height:40px; border-radius:50%; background:#fff; color:#000; display:flex; align-items:center; justify-content:center;"><i data-lucide="play" style="width:18px; height:18px; fill:#000; color:#000;"></i></button>
            <button class="btn-next-track"><i data-lucide="skip-forward" style="width:16px; height:16px;"></i></button>
            <button class="btn-shuffle-track" style="color:var(--t3);"><i data-lucide="shuffle" style="width:14px; height:14px;"></i></button>
          </div>
          <div style="display:flex; align-items:center; gap:8px; width:100px;">
            <i data-lucide="volume-2" style="width:12px; height:12px; color:var(--t3);"></i>
            <input type="range" class="vol-slider" min="0" max="100" value="50" style="flex:1;" />
          </div>
        </main>
      </div>
    `;

    const listEl = container.querySelector('.list-container');
    const titleEl = container.querySelector('.title-el');
    const artistEl = container.querySelector('.artist-el');
    const canvas = container.querySelector('.visualizer');
    const canvasCtx = canvas.getContext('2d');
    const timeEl = container.querySelector('.time-el');
    const timeTot = container.querySelector('.time-tot');
    const seeker = container.querySelector('.seeker');
    const loopBtn = container.querySelector('.btn-loop-track');
    const prevBtn = container.querySelector('.btn-prev-track');
    const playBtn = container.querySelector('.btn-play-track');
    const nextBtn = container.querySelector('.btn-next-track');
    const shuffleBtn = container.querySelector('.btn-shuffle-track');
    const volSlider = container.querySelector('.vol-slider');
    const discIcon = container.querySelector('.disc-icon');

    function render() {
      listEl.innerHTML = '';
      tracks.forEach((t, i) => {
        const item = document.createElement('div');
        const active = i === activeIdx;
        item.style.cssText = `padding:6px 8px; border-radius:6px; cursor:pointer; background:${active ? 'rgba(255,255,255,0.06)' : 'transparent'}; font-size:11px;`;
        item.innerHTML = `<strong style="color:${active ? 'var(--t1)' : 'var(--t2)'}">${t.name}</strong><br/><span style="color:var(--t3); font-size:9px;">${t.artist}</span>`;
        item.addEventListener('click', () => selectTrack(i));
        listEl.appendChild(item);
      });
    }

    function selectTrack(i) {
      stop();
      activeIdx = i;
      const track = tracks[activeIdx];
      titleEl.innerText = track.name;
      artistEl.innerText = track.artist;
      elapsed = 0;
      seeker.value = 0;
      timeEl.innerText = '0:00';
      timeTot.innerText = formatTime(track.dur);
      render();
      if (state) start();
    }

    function formatTime(s) {
      const min = Math.floor(s / 60);
      const sec = Math.floor(s % 60);
      return `${min}:${sec < 10 ? '0' : ''}${sec}`;
    }

    function start() {
      if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        gainNode = audioCtx.createGain();
        analyserNode = audioCtx.createAnalyser();
        analyserNode.fftSize = 64;
        gainNode.connect(analyserNode);
        analyserNode.connect(audioCtx.destination);
      }
      if (audioCtx.state === 'suspended') audioCtx.resume();

      const track = tracks[activeIdx];
      osc = audioCtx.createOscillator();
      const freq = { 'A4': 440, 'C5': 523.25, 'E4': 329.63 }[track.note] || 440;
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
      osc.connect(gainNode);
      gainNode.gain.setValueAtTime(volume / 100 * 0.1, audioCtx.currentTime);
      osc.start();

      draw();

      timer = setInterval(() => {
        if (elapsed < track.dur) {
          elapsed++;
          seeker.value = (elapsed / track.dur) * 100;
          timeEl.innerText = formatTime(elapsed);
        } else {
          if (loop) selectTrack(activeIdx);
          else next();
        }
      }, 1000);

      discIcon.style.animation = 'spin 6s linear infinite';
    }

    function stop() {
      if (osc) {
        try { osc.stop(); osc.disconnect(); } catch (e) {}
        osc = null;
      }
      if (timer) { clearInterval(timer); timer = null; }
      cancelAnimationFrame(animId);
      discIcon.style.animation = 'none';
      canvasCtx.clearRect(0, 0, canvas.width, canvas.height);
    }

    function next() {
      let idx = activeIdx + 1;
      if (shuffle) idx = Math.floor(Math.random() * tracks.length);
      else if (idx >= tracks.length) idx = 0;
      selectTrack(idx);
    }

    function prev() {
      let idx = activeIdx - 1;
      if (idx < 0) idx = tracks.length - 1;
      selectTrack(idx);
    }

    function draw() {
      animId = requestAnimationFrame(draw);
      const len = analyserNode.frequencyBinCount;
      const data = new Uint8Array(len);
      analyserNode.getByteTimeDomainData(data);

      canvasCtx.clearRect(0, 0, canvas.width, canvas.height);
      canvasCtx.lineWidth = 1.5;
      canvasCtx.strokeStyle = 'rgba(255, 255, 255, 0.3)';
      canvasCtx.beginPath();

      const slice = canvas.width / len;
      let x = 0;
      for (let i = 0; i < len; i++) {
        const v = data[i] / 128.0;
        const y = v * canvas.height / 2;
        if (i === 0) canvasCtx.moveTo(x, y);
        else canvasCtx.lineTo(x, y);
        x += slice;
      }
      canvasCtx.lineTo(canvas.width, canvas.height / 2);
      canvasCtx.stroke();
    }

    playBtn.addEventListener('click', () => {
      state = !state;
      if (state) {
        playBtn.innerHTML = '<i data-lucide="pause" style="width:18px; height:18px; fill:#000; color:#000;"></i>';
        start();
      } else {
        playBtn.innerHTML = '<i data-lucide="play" style="width:18px; height:18px; fill:#000; color:#000;"></i>';
        stop();
      }
      if (typeof lucide !== 'undefined') lucide.createIcons({ attrs: { class: 'lucide-icon' } });
    });

    nextBtn.addEventListener('click', next);
    prevBtn.addEventListener('click', prev);

    loopBtn.addEventListener('click', () => {
      loop = !loop;
      loopBtn.style.color = loop ? 'var(--t1)' : 'var(--t3)';
    });

    shuffleBtn.addEventListener('click', () => {
      shuffle = !shuffle;
      shuffleBtn.style.color = shuffle ? 'var(--t1)' : 'var(--t3)';
    });

    volSlider.addEventListener('input', (e) => {
      volume = e.target.value;
      if (gainNode) gainNode.gain.setValueAtTime(volume / 100 * 0.15, audioCtx.currentTime);
    });

    seeker.addEventListener('input', (e) => {
      const track = tracks[activeIdx];
      elapsed = Math.floor((e.target.value / 100) * track.dur);
      timeEl.innerText = formatTime(elapsed);
    });

    const observer = new MutationObserver(() => {
      if (!document.body.contains(container)) {
        stop();
        if (audioCtx) audioCtx.close();
        observer.disconnect();
      }
    });
    observer.observe(document.body, { childList: true, subtree: true });

    render();
    selectTrack(0);
  }
});
